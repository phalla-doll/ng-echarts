Browser Share

Visitors by browser across five categories.

Standalone Angular component
import { ChangeDetectionStrategy, Component } from '@angular/core';
import { PieChart, type ChartMeasure } from '@ng-echarts/chart-library/pie';
interface Browser { id: string; browser: string; visitors: number | null; }
@Component({
  selector: 'app-browser-chart', imports: [PieChart],
  changeDetection: ChangeDetectionStrategy.OnPush,
  template: `<nec-pie-chart title="Browser Share" description="Visitors by browser across five categories."
    [data]="rows" rowIdKey="id" categoryKey="browser" [measure]="measure"
    [innerRadius]="0" [showLabels]="false"
    theme="dark" [showDataTable]="false" [height]="300"
    [style.--nec-series-1]="'#60a5fa'" [style.--nec-series-2]="'#34d399'" [style.--nec-series-3]="'#fbbf24'" [style.--nec-series-4]="'#a78bfa'" [style.--nec-series-5]="'#9ca3af'"
  />`,
})
export class BrowserChart {
  readonly rows: readonly Browser[] = [
  {
    "id": "chrome",
    "browser": "Chrome",
    "visitors": 275
  },
  {
    "id": "safari",
    "browser": "Safari",
    "visitors": 200
  },
  {
    "id": "firefox",
    "browser": "Firefox",
    "visitors": 187
  },
  {
    "id": "edge",
    "browser": "Edge",
    "visitors": 173
  },
  {
    "id": "other",
    "browser": "Other",
    "visitors": 90
  }
];
  readonly measure: ChartMeasure<Browser> = {"id":"visitors","key":"visitors","label":"Visitors"};
}
